# DUO LUDO  v0.1

2 players ka board game. Humo (human vs human) aur Robo (human vs robot).

## Kaise chalayein (sabse aasan)
1. Zip ko **Extract / Unzip** karo (right-click -> Extract All).
2. Folder me `duo-ludo.html` file double-click karo.
3. Chrome / Edge / Safari me khul jayega. Bas. Internet ki zaroorat nahi.

(Ya poora project chalana ho to `index.html` kholo, same cheez hai.)

## Game ke rules
- Token ko board pe laane ke liye **1** chahiye.
- **6**, capture, ya token home pahunchne par **extra turn**.
- Home pe pahunchne ke liye **exact** number chahiye.
- Star = safe tile. Bolt = +3 steps. Do token ek tile pe = protected.
- Gate (arrow wala tile) se token apni home lane me mudta hai, start tile se alag.

## Folder
- `index.html`, `css/`, `js/` : source code
- `tools/build.py` : single-file build (`python3 tools/build.py`)
- `tests/engine.test.js` : rules/fairness test (`node tests/engine.test.js`)

## Online launch
`dist/duo-ludo.html` ko kisi bhi static hosting (Netlify, GitHub Pages, Vercel) pe upload kar do.

(function (DL) {
  'use strict';

  const TAU = Math.PI * 2;
  const LANE = DL.config.lane;

  /* ---- outlines: closed clockwise paths, first point = player one's tip ---- */

  function ellipse(a, b) {
    const out = [];
    for (let i = 0; i < 4000; i++) {
      const t = (i / 4000) * TAU;
      out.push({ x: -a * Math.cos(t), y: -b * Math.sin(t) });
    }
    return out;
  }

  function polygon(verts) {
    const out = [];
    for (let i = 0; i < verts.length; i++) {
      const a = verts[i];
      const b = verts[(i + 1) % verts.length];
      const steps = Math.max(2, Math.ceil(Math.hypot(b.x - a.x, b.y - a.y) / 1.5));
      for (let k = 0; k < steps; k++) {
        const f = k / steps;
        out.push({ x: a.x + (b.x - a.x) * f, y: a.y + (b.y - a.y) * f });
      }
    }
    return out;
  }

  function lemniscate(a, h) {
    const out = [];
    for (let i = 0; i < 4000; i++) {
      const t = (i / 4000) * TAU;
      out.push({ x: -a * Math.cos(t), y: (h / 2) * Math.sin(2 * t) });
    }
    return out;
  }

  const SQ3 = Math.sqrt(3) / 2;
  const R = 330;

  const DEFS = [
    {
      id: 'orbit', name: 'Orbit', tile: 'circle', n: 40, gate: 10, start: 0,
      blurb: 'A clean circle with round tiles. Smooth, fast and easy to read.',
      outline: () => ellipse(330, 240)
    },

      id: 'citadel', name: 'Citadel', tile: 'square', n: 48, gate: 12, start: 0,
      blurb: 'Square tiles along a walled rectangle. Long straights and sharp corners.',
      outline: () => polygon([{ x: -322, y: 0 }, { x: -322, y: -230 }, { x: 322, y: -230 }, { x: 322, y: 230 }, { x: -322, y: 230 }])
    },
    {
      id: 'prism', name: 'Prism', tile: 'triangle', n: 48, gate: 12, start: 0,
      blurb: 'A hexagonal ring of triangles that always point the way forward.',
      outline: () => polygon([
        { x: -R, y: 0 }, { x: -R / 2, y: -R * SQ3 }, { x: R / 2, y: -R * SQ3 },
        { x: R, y: 0 }, { x: R / 2, y: R * SQ3 }, { x: -R / 2, y: R * SQ3 }
      ])
    },
    {
      id: 'rhombus', name: 'Rhombus', tile: 'diamond', n: 40, gate: 10, start: 0,
      blurb: 'Diamond tiles on a wide rhombus. The gates sit at the sharp points.',
      outline: () => polygon([{ x: -400, y: 0 }, { x: 0, y: -240 }, { x: 400, y: 0 }, { x: 0, y: 240 }])
    },
    {
      id: 'infinity', name: 'Infinity', tile: 'circle', n: 40, gate: 5, start: 5, junction: [10, 30],
      blurb: 'A figure eight with a crossroads in the middle. Everyone passes twice.',
      outline: () => lemniscate(330, 320)
    }

  function resample(dense, n) {
    const m = dense.length;
    const cum = [0];
    for (let i = 1; i <= m; i++) {
      const a = dense[i - 1];
      const b = dense[i % m];
      cum.push(cum[i - 1] + Math.hypot(b.x - a.x, b.y - a.y));
    }
    const total = cum[m];
    const pts = [];
    let j = 0;
    for (let k = 0; k < n; k++) {
      const target = (k * total) / n;
      while (j < m - 1 && cum[j + 1] < target) j++;
      const a = dense[j];
      const b = dense[(j + 1) % m];
      const f = (target - cum[j]) / (cum[j + 1] - cum[j] || 1);
      pts.push({ x: a.x + (b.x - a.x) * f, y: a.y + (b.y - a.y) * f });
    }
    return { pts, total };
  }

  function build(id, portrait) {
    const def = DEFS.find((d) => d.id === id) || DEFS[0];
    let dense = def.outline();
    if (portrait) dense = dense.map((p) => ({ x: p.y, y: -p.x }));

    const { pts, total } = resample(dense, def.n);
    const n = def.n;
    const spacing = total / n;
    const tileId = Array.from({ length: n }, (_, i) => i);
    const junction = new Set();
    (def.junction || []).forEach((i) => {
      pts[i] = { x: 0, y: 0 };
      junction.add(i);
    });
    if (def.junction) tileId[def.junction[1]] = def.junction[0];

    const ang = pts.map((_, i) => {
      const a = pts[(i - 1 + n) % n];
      const b = pts[(i + 1) % n];
      return Math.atan2(b.y - a.y, b.x - a.x);
    });

    const s0 = def.start || 0;
    const start = [s0, (s0 + n / 2) % n];
    const entry = start.map((s) => (s - def.gate + n) % n);
    const rl = n - def.gate + 1;
    const len = rl + LANE;

    const lanes = entry.map((e) => {
      const p = pts[e];
      const dist = Math.hypot(p.x, p.y);
      const dir = { x: -p.x / dist, y: -p.y / dist };
      const step = Math.min(spacing * 1.15, (0.8 * dist) / LANE);
      return Array.from({ length: LANE }, (_, i) => ({
        x: p.x + dir.x * step * (i + 1),
        y: p.y + dir.y * step * (i + 1)
      }));
    });
    const laneDir = entry.map((e, p) => {
      const a = pts[e];
      const b = lanes[p][0];
      return Math.atan2(b.y - a.y, b.x - a.x);
    });

    const a = Math.round(n * 0.2);
    const b = Math.round(n * 0.3);
    const safe = new Set([0, n / 2, a, b, a + n / 2, b + n / 2].map((r) => (s0 + r) % n));
    const bo = Math.round(n * 0.075);
    const bp = Math.round(n * 0.425);
    const boost = new Set([bo, bp, bo + n / 2, bp + n / 2].map((r) => (s0 + r) % n));

    const tileR = spacing * 0.36;
    let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
    dense.forEach((p) => {
      if (p.x < minX) minX = p.x;
      if (p.x > maxX) maxX = p.x;
      if (p.y < minY) minY = p.y;
      if (p.y > maxY) maxY = p.y;
    });
    const pad = tileR * 2.2;

    return {
      id: def.id, name: def.name, blurb: def.blurb, tile: def.tile, portrait: !!portrait,
      n, gate: def.gate, rl, len, home: len - 1, lane: LANE,
      pts, ang, tileId, junction, start, entry, lanes, laneDir, safe, boost,
      spacing, tileR, dense,
      bounds: { minX: minX - pad, maxX: maxX + pad, minY: minY - pad, maxY: maxY + pad }
    };
  }

  DL.Maps = {
    list: DEFS.map((d) => ({ id: d.id, name: d.name, blurb: d.blurb, tile: d.tile })),
    build
  };
})(globalThis.DL = globalThis.DL || {});

(function (DL) {
  'use strict';

  const clamp = (x, a, b) => Math.max(a, Math.min(b, x));

  function mulberry32(a) {
    return function () {
      a |= 0; a = (a + 0x6D2B79F5) | 0;
      let t = Math.imul(a ^ (a >>> 15), 1 | a);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  const ringIdx = (M, p, prog) => (M.start[p] + prog) % M.n;
  const posOf = (M, p, prog) => (prog < M.rl ? M.pts[ringIdx(M, p, prog)] : M.lanes[p][prog - M.rl]);
  const tileKey = (M, p, prog) => (prog < M.rl ? 'r' + M.tileId[ringIdx(M, p, prog)] : 'l' + p + '_' + (prog - M.rl));

  function create(o) {
    const mk = (name, bot) => ({
      name, bot,
      tokens: Array(o.tokens).fill(-1),
      rolls: [], misses: 0, homeMisses: 0,
      stats: { rolls: 0, sixes: 0, captures: 0, extra: 0, moves: 0, lost: 0 }
    });
    return {
      map: DL.Maps.build(o.map, !!o.portrait),
      portrait: !!o.portrait,
      n: o.tokens,
      level: o.level || 'normal',
      turn: 0,
      moves: 0,
      rng: mulberry32(o.seed || ((Date.now() ^ (Math.random() * 1e9)) >>> 0)),
      players: [mk(o.names[0], false), mk(o.names[1], !!o.bot)]
    };
  }

  function legalMoves(st, p, v) {
    const M = st.map;
    const out = [];
    const seen = new Set();
    st.players[p].tokens.forEach((t, i) => {
      if (t === M.home) return;
      if (t < 0) { if (v !== 1) return; }
      else if (t + v > M.home) return;
      if (seen.has(t)) return;
      seen.add(t);
      out.push(i);
    });
    return out;
  }

  function oppOnTile(st, p, prog) {
    const M = st.map;
    const o = 1 - p;
    const tid = M.tileId[ringIdx(M, p, prog)];
    const res = [];
    st.players[o].tokens.forEach((t, i) => {
      if (t >= 0 && t < M.rl && M.tileId[ringIdx(M, o, t)] === tid) res.push(i);
    });
    return res;
  }

  function resolveCapture(st, p, q, caps, idx) {
    const M = st.map;
    if (q >= M.rl) return;
    if (M.safe.has(ringIdx(M, p, q))) return;
    const list = oppOnTile(st, p, q);
    if (list.length !== 1) return;
    const o = 1 - p;
    const ti = list[0];
    caps.push({ p: o, ti, prog: st.players[o].tokens[ti], idx });
    st.players[o].tokens[ti] = -1;
    st.players[p].stats.captures++;
    st.players[o].stats.lost++;
  }

  function doMove(st, p, ti, v) {
    const M = st.map;
    const pl = st.players[p];
    const from = pl.tokens[ti];
    let q = from < 0 ? 0 : from + v;
    const seq = [];
    if (from < 0) seq.push(0);
    else for (let k = from + 1; k <= q; k++) seq.push(k);
    pl.tokens[ti] = q;

    const caps = [];
    resolveCapture(st, p, q, caps, seq.length - 1);

    let boosted = false;
    if (q < M.rl && M.boost.has(ringIdx(M, p, q)) && q + 3 <= M.rl - 1) {
      boosted = true;
      for (let k = q + 1; k <= q + 3; k++) seq.push(k);
      q += 3;
      pl.tokens[ti] = q;
      resolveCapture(st, p, q, caps, seq.length - 1);
    }
    if (q === M.home) pl.homeMisses = 0;
    pl.stats.moves++;
    st.moves++;
    return { p, ti, from, to: q, seq, caps, boosted, entered: from < 0, reachedHome: q === M.home, captured: caps.length > 0 };
  }

  const isWin = (st, p) => st.players[p].tokens.every((t) => t === st.map.home);

  /* Fair Drama Engine: both players share identical rules; adjustments are bounded. */
  function rollDice(st, p) {
    const M = st.map;
    const me = st.players[p];
    const op = st.players[1 - p];
    const w = [0, 1, 1, 1, 1, 1, 1];
    const reasons = [];

    if (me.tokens.every((t) => t < 0) && me.misses > 0) {
      w[1] *= 1 + 0.22 * Math.min(me.misses, 6);
      reasons.push('Helping you leave base');
    }

    let need = 0;
    for (const t of me.tokens) {
      if (t >= 0 && t < M.home && M.home - t <= 6) {
        const n = M.home - t;
        if (!need || n < need) need = n;
      }
    }
    if (need) {
      if (me.homeMisses >= 3) { w[need] *= 1 + 0.15 * Math.min(me.homeMisses - 2, 4); reasons.push('Home stretch nudge'); }
      else if (me.homeMisses <= 1) w[need] *= 0.9;
    }

    const prog = (s) => s.tokens.reduce((a, t) => a + (t < 0 ? 0 : t + 1), 0) / (st.n * M.len);
    const s = clamp((prog(op) - prog(me)) * 5, -1, 1);
    for (let k = 1; k <= 6; k++) w[k] *= k >= 4 ? 1 + 0.3 * s : 1 - 0.15 * s;
    if (s > 0.35) reasons.push('Catch-up boost');
    else if (s < -0.35) reasons.push('Leader balance');

    const r = me.rolls;
    if (r.length >= 2 && r[r.length - 1] === r[r.length - 2]) {
      w[r[r.length - 1]] *= 0.5;
      reasons.push('Streak breaker');
    }

    for (let k = 1; k <= 6; k++) {
      w[k] *= 0.9 + 0.2 * st.rng();
      w[k] = clamp(w[k], 0.5, 1.7);
    }
    let total = 0;
    for (let k = 1; k <= 6; k++) total += w[k];
    let x = st.rng() * total;
    let v = 6;
    for (let k = 1; k <= 6; k++) { x -= w[k]; if (x <= 0) { v = k; break; } }

    me.rolls.push(v);
    if (me.rolls.length > 6) me.rolls.shift();
    me.stats.rolls++;
    if (v === 6) me.stats.sixes++;

    const legal = legalMoves(st, p, v);
    me.misses = legal.length === 0 ? me.misses + 1 : 0;
    if (need) {
      if (legal.some((i) => me.tokens[i] >= 0 && me.tokens[i] + v === M.home)) me.homeMisses = 0;
      else me.homeMisses++;
    }
    return { v, w: w.slice(1), reasons, legal };
  }

  /* ---- robo ---- */

  function dangerAt(st, p, prog) {
    const M = st.map;
    if (prog < 0 || prog >= M.rl) return 0;
    const r = ringIdx(M, p, prog);
    if (M.safe.has(r)) return 0;
    const tid = M.tileId[r];
    const o = 1 - p;
    let d = 0;
    for (const t of st.players[o].tokens) {
      if (t < 0 || t >= M.rl) continue;
      const ro = ringIdx(M, o, t);
      for (let k = 1; k <= 6; k++) {
        if (t + k <= M.rl - 1 && M.tileId[(ro + k) % M.n] === tid) { d++; break; }
      }
    }
    return d;
  }

  function scoreMove(st, p, ti, v) {
    const M = st.map;
    const t = st.players[p].tokens[ti];
    const q = t < 0 ? 0 : t + v;
    let s = 0;
    if (q === M.home) s += 100;
    else if (q >= M.rl) s += 26 + (q - M.rl) * 2;
    if (q < M.rl) {
      const r = ringIdx(M, p, q);
      if (M.safe.has(r)) s += 20;
      else {
        const c = oppOnTile(st, p, q);
        if (c.length === 1) s += 55 + st.players[1 - p].tokens[c[0]] * 0.6;
      }
      if (M.boost.has(r) && q + 3 <= M.rl - 1) s += 16;
    }
    if (t < 0) s += 34;
    s += q * 0.35;
    s += ((t >= 0 ? dangerAt(st, p, t) : 0) - dangerAt(st, p, q)) * 18;
    return s;
  }

  function botChoose(st, p, v, legal) {
    const lv = DL.config.levels[st.level] || DL.config.levels.normal;
    if (lv.random) return legal[Math.floor(st.rng() * legal.length)];
    let best = legal[0];
    let bs = -Infinity;
    for (const ti of legal) {
      const sc = scoreMove(st, p, ti, v) + st.rng() * lv.noise;
      if (sc > bs) { bs = sc; best = ti; }
    }
    return best;
  }

  DL.Engine = { create, legalMoves, doMove, rollDice, botChoose, isWin, ringIdx, posOf, tileKey };
})(globalThis.DL = globalThis.DL || {});

  function drawToken(c, x, y, r, col, o) {
    o = o || {};
    const R = r * (o.s || 1);
    const hop = o.hop || 0;
    c.save();
    c.fillStyle = 'rgba(38,43,66,' + clamp(0.22 - hop * 0.004, 0.06, 0.22) + ')';
    c.beginPath(); c.ellipse(x, y + R * 0.62, R * 0.9, R * 0.34, 0, 0, TAU); c.fill();

    const cy = y - hop - R * 0.12;
    if (o.glow) {
      const pu = (Math.sin(o.t / 170) + 1) / 2;
      c.lineWidth = 2.5;
      c.strokeStyle = alpha(col.main, 0.55 + pu * 0.35);
      c.beginPath(); c.arc(x, cy, R + 4 + pu * 3, 0, TAU); c.stroke();
    }
    // base
    c.fillStyle = col.deep;
    c.beginPath(); c.ellipse(x, cy + R * 0.5, R * 0.92, R * 0.4, 0, 0, TAU); c.fill();
    c.fillStyle = mix(col.deep, '#FFFFFF', 0.25);
    c.beginPath(); c.ellipse(x, cy + R * 0.42, R * 0.92, R * 0.36, 0, 0, TAU); c.fill();
    // body
    const g = c.createRadialGradient(x - R * 0.3, cy - R * 0.35, R * 0.05, x, cy, R * 0.95);
    g.addColorStop(0, '#FFFFFF');
    g.addColorStop(0.28, col.light);
    g.addColorStop(0.62, col.main);
    g.addColorStop(1, col.deep);
    c.fillStyle = g;
    c.beginPath(); c.arc(x, cy, R * 0.82, 0, TAU); c.fill();
    c.lineWidth = Math.max(1, R * 0.06);
    c.strokeStyle = alpha(col.deep, 0.55);
    c.stroke();
    c.fillStyle = 'rgba(255,255,255,0.85)';
    c.beginPath(); c.ellipse(x - R * 0.28, cy - R * 0.42, R * 0.22, R * 0.12, -0.6, 0, TAU); c.fill();
    c.restore();
  }

  let host = null, cv = null, ctx = null;
  let S = null;
  let W = 0, H = 0, dpr = 1;
  let L = { ready: false };
  let disp = {};
  let fx = [];
  let rest = {};
  let hooks = {};
  let last = 0;
  let dockSize = 56;
  let tr = 14;
  const ui = { turn: 0, phase: 'idle', legal: null, v: 0, hint: '', canRoll: false };
  const dice = { x: 0, y: 0, rot: 0, hop: 0, face: 1, owner: 0, moving: false, rolling: false, sx: 1, sy: 1 };
  const PIPS = { 1: [4], 2: [0, 8], 3: [0, 4, 8], 4: [0, 2, 6, 8], 5: [0, 2, 4, 6, 8], 6: [0, 2, 3, 5, 6, 8] };

  const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
  const easeIO = (k) => (k < 0.5 ? 2 * k * k : 1 - Math.pow(-2 * k + 2, 2) / 2);

  function tween(o, to, ms, hopH) {
    return new Promise((res) => {
      const sx = o.x, sy = o.y, t0 = performance.now();
      function step(t) {
        const k = Math.min(1, (t - t0) / ms);
        const e = easeIO(k);
        o.x = sx + (to.x - sx) * e;
        o.y = sy + (to.y - sy) * e;
        o.hop = hopH ? Math.sin(Math.PI * k) * hopH : 0;
        if (k < 1) requestAnimationFrame(step);
        else { o.x = to.x; o.y = to.y; o.hop = 0; res(); }
      }
      requestAnimationFrame(step);
    });
  }

  function init(canvas, h) {
    cv = canvas;
    ctx = cv.getContext('2d');
    host = cv.parentElement;
    hooks = h || {};
    if (globalThis.ResizeObserver) new ResizeObserver(() => layout()).observe(host);
    globalThis.addEventListener('resize', layout);
    cv.addEventListener('pointerdown', onDown);
    cv.addEventListener('pointermove', onMove);
    requestAnimationFrame(frame);
  }

  function layout() {
    if (!host) return;
    const rect = host.getBoundingClientRect();
    if (rect.width < 40 || rect.height < 40) { L.ready = false; return; }
    W = Math.floor(rect.width); H = Math.floor(rect.height);
    dpr = Math.min(globalThis.devicePixelRatio || 1, 2.5);
    cv.width = Math.round(W * dpr); cv.height = Math.round(H * dpr);
    cv.style.width = W + 'px'; cv.style.height = H + 'px';
    if (!S) return;

    const portrait = H > W;
    if (S.portrait !== portrait) { S.map = DL.Maps.build(S.map.id, portrait); S.portrait = portrait; }
    const M = S.map;
    const pad = clamp(Math.min(W, H) * 0.022, 8, 18);
    const stripH = H < 430 ? 54 : clamp(H * 0.125, 64, 98);
    const areaX = pad;
    const areaY = pad + stripH + pad * 0.7;
    const areaW = W - pad * 2;
    const areaH = Math.max(60, H - 2 * (pad + stripH) - pad * 1.4);
    const bw = M.bounds.maxX - M.bounds.minX;
    const bh = M.bounds.maxY - M.bounds.minY;
    const s = Math.min(areaW / bw, areaH / bh);
    const ox = areaX + areaW / 2 - ((M.bounds.minX + M.bounds.maxX) / 2) * s;
    const oy = areaY + areaH / 2 - ((M.bounds.minY + M.bounds.maxY) / 2) * s;
    tr = clamp(M.tileR * s * 0.86, 6, stripH * 0.24);
    dockSize = clamp(stripH * 0.72, 40, 68);

    const strip = [
      { x: pad, y: H - pad - stripH, w: W - pad * 2, h: stripH },
      { x: pad, y: pad, w: W - pad * 2, h: stripH }
    ];
    const dock = [
      { x: strip[0].x + strip[0].w - 14 - dockSize / 2, y: strip[0].y + strip[0].h / 2 },
      { x: strip[1].x + 14 + dockSize / 2, y: strip[1].y + strip[1].h / 2 }
    ];
    const rowMax = W - pad * 2 - 32 - dockSize * 1.7;
    const spacing = Math.min(tr * 2.5, rowMax / S.n);
    L = {
      ready: true, pad, stripH, strip, dock, s, ox, oy, spacing, portrait,
      rowY: [strip[0].y + strip[0].h * 0.68, strip[1].y + strip[1].h * 0.32],
      nameY: [strip[0].y + strip[0].h * 0.3, strip[1].y + strip[1].h * 0.7]
    };

    const cache = document.createElement('canvas');
    cache.width = cv.width; cache.height = cv.height;
    const cc = cache.getContext('2d');
    cc.setTransform(dpr, 0, 0, dpr, 0, 0);
    drawStatic(cc, M, { s, ox, oy }, false);
    L.cache = cache;

    if (!dice.moving) { dice.x = dock[dice.owner].x; dice.y = dock[dice.owner].y; }
  }

  const pos = (p, prog) => { const q = E.posOf(S.map, p, prog); return { x: L.ox + q.x * L.s, y: L.oy + q.y * L.s }; };
  function slot(p, i) {
    const R = L.strip[p];
    return p === 0
      ? { x: R.x + 18 + tr + i * L.spacing, y: L.rowY[0] }
      : { x: R.x + R.w - 18 - tr - i * L.spacing, y: L.rowY[1] };
  }

  /* ---- effects ---- */
  function burstFx(x, y, color, n, speed) {
    for (let i = 0; i < n; i++) {
      const a = Math.random() * TAU;
      const sp = speed * (0.35 + Math.random() * 0.8);
      fx.push({ k: 'dot', x, y, vx: Math.cos(a) * sp, vy: Math.sin(a) * sp - speed * 0.2, t: 0, max: 0.45 + Math.random() * 0.45, size: 2 + Math.random() * 3, color });
    }
  }
  function ripple(x, y, color, r1) { fx.push({ k: 'ring', x, y, t: 0, max: 0.5, r1, color }); }

  function drawFx(dt) {
    fx = fx.filter((f) => (f.t += dt) < f.max);
    fx.forEach((f) => {
      const k = f.t / f.max;
      if (f.k === 'ring') {
        ctx.strokeStyle = alpha(f.color, (1 - k) * 0.6);
        ctx.lineWidth = 2 * (1 - k) + 0.5;
        ctx.beginPath(); ctx.arc(f.x, f.y, f.r1 * (0.3 + k * 0.9), 0, TAU); ctx.stroke();
      } else {
        f.x += f.vx * dt; f.y += f.vy * dt; f.vy += 420 * dt;
        ctx.fillStyle = alpha(f.color, 1 - k);
        ctx.beginPath(); ctx.arc(f.x, f.y, f.size * (1 - k * 0.5), 0, TAU); ctx.fill();
      }
    });
  }

  /* ---- frame ---- */
  function frame(now) {
    const dt = Math.min(0.05, (now - last) / 1000 || 0.016);
    last = now;
    draw(now, dt);
    requestAnimationFrame(frame);
  }

  function computeRest() {
    const groups = {};
    const res = {};
    S.players.forEach((pl, p) => pl.tokens.forEach((t, ti) => {
      const key = p * 10 + ti;
      if (disp[key]) return;
      if (t < 0) { const q = slot(p, ti); res[key] = { x: q.x, y: q.y, s: 1 }; return; }
      const k = E.tileKey(S.map, p, t);
      (groups[k] = groups[k] || []).push([p, ti, t]);
    }));
    for (const k in groups) {
      const arr = groups[k];
      const n = arr.length;
      const base = pos(arr[0][0], arr[0][2]);
      arr.forEach(([p, ti], i) => {
        let ox = 0, oy = 0, sc = 1;
        if (n === 2) { ox = (i ? 1 : -1) * tr * 0.55; sc = 0.84; }
        else if (n > 2) {
          const a = (i / n) * TAU + 0.5;
          const rad = tr * (0.45 + n * 0.1);
          ox = Math.cos(a) * rad; oy = Math.sin(a) * rad; sc = 0.7;
        }
        res[p * 10 + ti] = { x: base.x + ox, y: base.y + oy, s: sc };
      });
    }
    return res;
  }

  function stripLabel(p, now) {
    const pl = S.players[p];
    const R = L.strip[p];
    const active = ui.turn === p && ui.phase !== 'over' && ui.phase !== 'idle';
    const fs = clamp(L.stripH * 0.2, 12, 17);
    const home = pl.tokens.filter((t) => t === S.map.home).length;
    const tail = active && ui.hint ? ui.hint : 'Home ' + home + '/' + S.n + '   Captures ' + pl.stats.captures;
    const y = L.nameY[p];
    ctx.textBaseline = 'middle';

    ctx.font = '700 ' + fs + 'px ' + FONT;
    const name = pl.name + (pl.bot ? '  ROBO' : '');
    const nw = ctx.measureText(name).width;
    ctx.font = '500 ' + fs * 0.84 + 'px ' + FONT;
    const tw = ctx.measureText(tail).width;
    const dotR = fs * 0.3;
    const gap = fs * 0.8;
    const roomW = R.w - 28 - dockSize * 1.4;
    const showTail = nw + dotR * 2 + gap + tw < roomW;

    const draw = (x, align) => {
      ctx.textAlign = align;
      ctx.font = '700 ' + fs + 'px ' + FONT;
      ctx.fillStyle = P.ink;
      ctx.fillText(name, x, y);
    };
    if (p === 0) {
      let x = R.x + 18;
      ctx.fillStyle = COL[p].main; ctx.beginPath(); ctx.arc(x + dotR, y, dotR, 0, TAU); ctx.fill();
      x += dotR * 2 + fs * 0.5;
      draw(x, 'left');
      x += nw + gap;
      if (showTail) { ctx.textAlign = 'left'; ctx.font = '500 ' + fs * 0.84 + 'px ' + FONT; ctx.fillStyle = active && ui.hint ? COL[p].deep : P.muted; ctx.fillText(tail, x, y); }
    } else {
      let x = R.x + R.w - 18;
      ctx.fillStyle = COL[p].main; ctx.beginPath(); ctx.arc(x - dotR, y, dotR, 0, TAU); ctx.fill();
      x -= dotR * 2 + fs * 0.5;
      draw(x, 'right');
      x -= nw + gap;
      if (showTail) { ctx.textAlign = 'right'; ctx.font = '500 ' + fs * 0.84 + 'px ' + FONT; ctx.fillStyle = active && ui.hint ? COL[p].deep : P.muted; ctx.fillText(tail, x, y); }
    }
  }

  function drawStrips(now) {
    [0, 1].forEach((p) => {
      const R = L.strip[p];
      const active = ui.turn === p && ui.phase !== 'over' && ui.phase !== 'idle';
      ctx.save();
      if (active) { ctx.shadowColor = alpha(COL[p].main, 0.42); ctx.shadowBlur = 20; ctx.shadowOffsetY = 4; }
      else { ctx.shadowColor = 'rgba(70,95,170,0.10)'; ctx.shadowBlur = 10; ctx.shadowOffsetY = 3; }
      rr(ctx, R.x, R.y, R.w, R.h, R.h * 0.3);
      ctx.fillStyle = active ? 'rgba(255,255,255,0.97)' : 'rgba(255,255,255,0.72)';
      ctx.fill();
      ctx.restore();
      ctx.lineWidth = active ? 1.8 : 1;
      ctx.strokeStyle = active ? COL[p].main : '#E2E8F6';
      rr(ctx, R.x, R.y, R.w, R.h, R.h * 0.3); ctx.stroke();

      // token row slots
      for (let i = 0; i < S.n; i++) {
        const q = slot(p, i);
        ctx.beginPath(); ctx.arc(q.x, q.y, tr * 0.95, 0, TAU);
        ctx.fillStyle = alpha(COL[p].main, 0.1); ctx.fill();
        ctx.lineWidth = 1.2; ctx.strokeStyle = alpha(COL[p].main, 0.35); ctx.setLineDash([3, 3]); ctx.stroke(); ctx.setLineDash([]);
      }
      // dice dock
      const d = L.dock[p];
      const ds = dockSize * 1.24;
      rr(ctx, d.x - ds / 2, d.y - ds / 2, ds, ds, ds * 0.26);
      ctx.fillStyle = alpha(COL[p].main, active ? 0.16 : 0.08); ctx.fill();
      ctx.lineWidth = 1.2; ctx.strokeStyle = alpha(COL[p].main, active ? 0.6 : 0.28); ctx.setLineDash([4, 4]); ctx.stroke(); ctx.setLineDash([]);
      if (active && ui.canRoll) {
        const pu = (Math.sin(now / 260) + 1) / 2;
        ctx.lineWidth = 2.5; ctx.strokeStyle = alpha(COL[p].main, 0.3 + pu * 0.5);
        rr(ctx, d.x - ds / 2 - 4 - pu * 3, d.y - ds / 2 - 4 - pu * 3, ds + 8 + pu * 6, ds + 8 + pu * 6, ds * 0.3); ctx.stroke();
      }
      stripLabel(p, now);
    });
  }

  function drawDice(now) {
    const size = dockSize;
    const ready = ui.canRoll && !dice.rolling && !dice.moving;
    const bob = ready ? Math.sin(now / 280) * size * 0.05 : 0;
    const x = dice.x, y = dice.y + bob;
    const hop = dice.hop;
    ctx.fillStyle = 'rgba(38,43,66,' + clamp(0.2 - hop / (size * 9), 0.05, 0.2) + ')';
    const sh = 1 - clamp(hop / (size * 4), 0, 0.5);
    ctx.beginPath(); ctx.ellipse(x, y + size * 0.52, size * 0.46 * sh, size * 0.13 * sh, 0, 0, TAU); ctx.fill();

    ctx.save();
    ctx.translate(x, y - hop);
    ctx.rotate(dice.rot);
    ctx.scale(dice.sx, dice.sy);
    const h = size / 2;
    ctx.save();
    ctx.shadowColor = 'rgba(70,95,170,0.28)'; ctx.shadowBlur = size * 0.2; ctx.shadowOffsetY = size * 0.08;
    rr(ctx, -h, -h, size, size, size * 0.24);
    const g = ctx.createLinearGradient(-h, -h, h, h);
    g.addColorStop(0, '#FFFFFF'); g.addColorStop(1, '#E7EDFA');
    ctx.fillStyle = g; ctx.fill();
    ctx.restore();
    ctx.lineWidth = 1.2; ctx.strokeStyle = '#D3DCF1';
    rr(ctx, -h, -h, size, size, size * 0.24); ctx.stroke();
    ctx.lineWidth = size * 0.03; ctx.strokeStyle = 'rgba(255,255,255,0.95)';
    rr(ctx, -h + size * 0.05, -h + size * 0.05, size * 0.9, size * 0.9, size * 0.2); ctx.stroke();
    const o = size * 0.25;
    (PIPS[dice.face] || PIPS[1]).forEach((i) => {
      const px = (i % 3 - 1) * o, py = (Math.floor(i / 3) - 1) * o;
      ctx.beginPath(); ctx.arc(px, py, size * 0.078, 0, TAU);
      ctx.fillStyle = dice.face === 1 ? COL[dice.owner].deep : P.ink; ctx.fill();
    });
    ctx.restore();
  }

  function draw(now, dt) {
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    ctx.clearRect(0, 0, W, H);
    if (!S || !L.ready) return;

    drawStrips(now);
    ctx.drawImage(L.cache, 0, 0, W, H);

    // legal move markers
    const legalProgs = ui.phase === 'pick' && ui.legal ? new Set(ui.legal.map((i) => S.players[ui.turn].tokens[i])) : null;
    if (legalProgs) {
      const pu = (Math.sin(now / 220) + 1) / 2;
      ui.legal.forEach((i) => {
        const tk = S.players[ui.turn].tokens[i];
        const q = pos(ui.turn, tk < 0 ? 0 : tk + ui.v);
        ctx.lineWidth = 2.2;
        ctx.setLineDash([5, 4]);
        ctx.strokeStyle = alpha(COL[ui.turn].deep, 0.6 + pu * 0.3);
        ctx.beginPath(); ctx.arc(q.x, q.y, tr * (1.25 + pu * 0.12), 0, TAU); ctx.stroke();
        ctx.setLineDash([]);
        ctx.fillStyle = alpha(COL[ui.turn].main, 0.18);
        ctx.fill();
      });
    }

    rest = computeRest();
    const items = [];
    S.players.forEach((pl, p) => pl.tokens.forEach((tk, ti) => {
      const key = p * 10 + ti;
      const d = disp[key];
      if (d) items.push({ p, ti, x: d.x, y: d.y, s: 1, hop: d.hop || 0, top: true });
      else if (rest[key]) items.push({ p, ti, x: rest[key].x, y: rest[key].y, s: rest[key].s, hop: 0 });
    }));
    items.sort((a, b) => (a.top ? 1 : 0) - (b.top ? 1 : 0) || a.y - b.y);
    items.forEach((it) => {
      const glow = legalProgs && it.p === ui.turn && legalProgs.has(S.players[it.p].tokens[it.ti]);
      drawToken(ctx, it.x, it.y, tr, COL[it.p], { s: it.s, hop: it.hop, glow, t: now });
    });

    drawFx(dt);
    drawDice(now);
  }

  /* ---- dice animation ---- */
  async function moveDiceTo(p, instant) {
    dice.owner = p;
    if (!L.ready) return;
    const t = L.dock[p];
    if (instant) { dice.x = t.x; dice.y = t.y; return; }
    dice.moving = true;
    SFX.whoosh();
    await tween(dice, t, 560, dockSize * 1.1);
    dice.moving = false;
    dice.x = L.dock[dice.owner].x; dice.y = L.dock[dice.owner].y;
  }

  async function rollDice(face, ms) {
    ms = ms || 950;
    dice.rolling = true;
    SFX.diceRoll(ms);
    const t0 = performance.now();
    const startRot = dice.rot;
    const spin = (Math.random() < 0.5 ? -1 : 1) * (2.4 + Math.random() * 1.4) * TAU;
    let lastFace = 0;
    await new Promise((res) => {
      function step(now) {
        const k = Math.min(1, (now - t0) / ms);
        const e = 1 - Math.pow(1 - k, 3);
        dice.rot = startRot + spin * e;
        dice.hop = Math.abs(Math.sin(k * Math.PI * 2.5)) * (1 - k) * dockSize * 0.95;
        dice.sx = 1 + 0.1 * Math.sin(k * Math.PI * 6) * (1 - k);
        dice.sy = 1 - 0.1 * Math.sin(k * Math.PI * 6) * (1 - k);
        if (now - lastFace > 45 + k * 120) { dice.face = 1 + Math.floor(Math.random() * 6); lastFace = now; }
        if (k < 1) requestAnimationFrame(step); else res();
      }
      requestAnimationFrame(step);
    });
    dice.face = face;
    dice.rot = Math.round(dice.rot / (Math.PI / 2)) * (Math.PI / 2);
    dice.hop = 0;
    dice.sx = 1.12; dice.sy = 0.88;
    await sleep(70);
    dice.sx = 0.97; dice.sy = 1.03;
    await sleep(70);
    dice.sx = 1; dice.sy = 1;
    dice.rolling = false;
    if (face === 6) SFX.six();
  }

  /* ---- token movement ---- */
  async function playMove(plan) {
    const key = plan.p * 10 + plan.ti;
    const col = COL[plan.p].main;
    const start = plan.from < 0 ? slot(plan.p, plan.ti) : pos(plan.p, plan.from);
    disp[key] = { x: start.x, y: start.y, hop: 0 };
    plan.caps.forEach((c) => { const q = pos(c.p, c.prog); disp[c.p * 10 + c.ti] = { x: q.x, y: q.y, hop: 0 }; });
    if (plan.entered) SFX.pop();

    let n = 0;
    for (let i = 0; i < plan.seq.length; i++) {
      const bs = plan.boosted && i >= plan.seq.length - 3;
      if (bs && i === plan.seq.length - 3) {
        SFX.boost();
        const q = pos(plan.p, plan.seq[i - 1]);
        burstFx(q.x, q.y, '#8E7CF0', 14, 150);
      }
      const to = pos(plan.p, plan.seq[i]);
      await tween(disp[key], to, bs ? 105 : 160, bs ? tr * 0.5 : tr * 0.95);
      SFX.step(n++);
      ripple(to.x, to.y, col, tr * 1.6);
      const cs = plan.caps.filter((c) => c.idx === i);
      if (cs.length) {
        SFX.capture();
        if (navigator.vibrate) navigator.vibrate(40);
        burstFx(to.x, to.y, COL[cs[0].p].main, 26, 230);
        ripple(to.x, to.y, '#FFFFFF', tr * 3);
        await Promise.all(cs.map(async (c) => {
          const k2 = c.p * 10 + c.ti;
          await tween(disp[k2], slot(c.p, c.ti), 620, tr * 3.2);
          delete disp[k2];
        }));
      }
    }
    SFX.land();
    if (plan.reachedHome) {
      const q = pos(plan.p, plan.to);
      SFX.home();
      burstFx(q.x, q.y, DL.config.palette.gold, 30, 220);
      burstFx(q.x, q.y, col, 18, 160);
    }
    delete disp[key];
  }

  /* ---- input ---- */
  function hit(px, py) {
    if (!S || !L.ready) return null;
    if (ui.canRoll && Math.hypot(px - dice.x, py - dice.y) < dockSize * 0.75) return { type: 'dice' };
    if (ui.phase === 'pick' && ui.legal) {
      const pl = S.players[ui.turn];
      const progs = new Set(ui.legal.map((i) => pl.tokens[i]));
      for (let ti = 0; ti < pl.tokens.length; ti++) {
        if (!progs.has(pl.tokens[ti])) continue;
        const r = rest[ui.turn * 10 + ti];
        if (r && Math.hypot(px - r.x, py - r.y) < tr * 1.5) return { type: 'token', ti };
      }
      for (const ti of ui.legal) {
        const t = pl.tokens[ti];
        const q = pos(ui.turn, t < 0 ? 0 : t + ui.v);
        if (Math.hypot(px - q.x, py - q.y) < tr * 1.5) return { type: 'token', ti };

(function (DL) {
  'use strict';

  const E = DL.Engine;

  const UI = DL.UI;
  const A = DL.Audio;

  const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

  let sid = 0;
  let busy = false;
  let paused = false;
  let waiters = [];
  let elapsed = 0;
  let tickStart = 0;
  let timerId = null;
  let lastRoll = null;
  let pendingRoll = null;

  const gate = () => (paused ? new Promise((r) => waiters.push(r)) : Promise.resolve());
  const alive = (id) => id === sid;
  const now = () => elapsed + (tickStart ? performance.now() - tickStart : 0);

  function startClock() {
    tickStart = performance.now();
    clearInterval(timerId);
    timerId = setInterval(() => { $('timer').textContent = UI.fmt(now()); }, 500);
  }
  function stopClock() {
    if (tickStart) elapsed += performance.now() - tickStart;
    tickStart = 0;
    clearInterval(timerId);
    $('timer').textContent = UI.fmt(elapsed);
  }

  function setPause(v) {
    paused = v;
    if (v) { stopClock(); UI.openModal('modalPause'); }
    else {
      UI.closeModal('modalPause');
      startClock();
      const w = waiters; waiters = [];
      w.forEach((f) => f());
    }
  }

  function start(c) {
    sid++;
    cfg = c;
    busy = false; paused = false; waiters = []; elapsed = 0; lastRoll = null;
    UI.closeModal('modalPause');
    UI.stopConfetti();
    A.stopMusic();
    UI.show('screen-game');
    const stage = $('stage').getBoundingClientRect();
    st = E.create({
      map: c.map, tokens: c.tokens, bot: c.mode === 'robo', level: c.level,
      names: c.names, portrait: stage.height > stage.width
    });
    $('mapPill').textContent = st.map.name;
    $('moves').textContent = '0';
    $('timer').textContent = '00:00';
    $('insight').hidden = true;
    R.setUI({ turn: 0, phase: 'idle', legal: null, v: 0, hint: '', canRoll: false });
    R.setState(st);
    // first player alternates by coin flip, fair for both
    st.turn = st.rng() < 0.5 ? 0 : 1;
    R.moveDiceTo(st.turn, true);
    startClock();
    const id = sid;
    sleep(650).then(() => { if (alive(id)) { UI.toast(st.players[st.turn].name + ' starts', 'soft'); beginTurn(id); } });
  }

  async function beginTurn(id) {
    await gate();
    if (!alive(id)) return;
    const p = st.turn;
    const bot = st.players[p].bot;
    busy = false;
    R.setUI({ turn: p, phase: 'roll', legal: null, v: 0, canRoll: !bot, hint: bot ? 'Thinking...' : 'Tap the dice' });
    if (bot) {
      await sleep(750);
      await gate();
      if (!alive(id)) return;
      doRoll(id);
    }
  }

  async function doRoll(id) {
    if (busy) return;
    busy = true;
    const p = st.turn;
    R.setUI({ phase: 'rolling', canRoll: false, hint: 'Rolling...' });
    const r = E.rollDice(st, p);
    lastRoll = { p, r };
    await R.rollDice(r.v, 950);
    if (!alive(id)) return;
    UI.insight(p, st, r);

    if (!r.legal.length) {
      A.sfx.nomove();
      UI.toast('No move for ' + r.v, 'soft');
      R.setUI({ hint: 'No move' });
      await sleep(900);
      await gate();
      if (!alive(id)) return;
      return endTurn(id, false);
    }

    const bot = st.players[p].bot;
    if (r.legal.length === 1 || bot) {
      R.setUI({ phase: 'moving', legal: null, hint: '' });
      await sleep(bot ? 600 : 300);
      await gate();
      if (!alive(id)) return;
      const ti = r.legal.length === 1 ? r.legal[0] : E.botChoose(st, p, r.v, r.legal);
      return move(id, ti, r.v);
    }
    pendingRoll = r;
    busy = false;
    R.setUI({ phase: 'pick', legal: r.legal, v: r.v, hint: 'Pick a token' });
  }

  async function move(id, ti, v) {
    busy = true;
    const p = st.turn;
    R.setUI({ phase: 'moving', legal: null, hint: '' });
    const plan = E.doMove(st, p, ti, v);
    $('moves').textContent = st.moves;
    await R.playMove(plan);
    if (!alive(id)) return;

    if (E.isWin(st, p)) return finish(id, p);

    const extra = v === 6 || plan.captured || plan.reachedHome;
    if (plan.captured) UI.toast('Captured!', 'hit');
    else if (plan.reachedHome) UI.toast('Home!', 'gold');
    else if (plan.boosted) UI.toast('Boost +3', '');
    if (extra) {
      st.players[p].stats.extra++;
      if (!plan.captured && !plan.reachedHome) UI.toast('Roll again', 'soft');
      A.sfx.extra();
    }
    await sleep(extra ? 650 : 350);
    await gate();
    if (!alive(id)) return;
    endTurn(id, extra);
  }

  async function endTurn(id, extra) {
    if (!extra) {
      st.turn = 1 - st.turn;
      R.setUI({ turn: st.turn, phase: 'idle', canRoll: false, hint: '' });
      await R.moveDiceTo(st.turn, false);

    }
  }

  function toMenu() {

  }

      }
    });
    $('btnSound').addEventListener('click', () => {

    });

    });
  }

    for (let i = 0; i < M.n; i++) for (let j = i + 1; j < M.n; j++) {
      if (M.tileId[i] === M.tileId[j]) continue;
      min = Math.min(min, Math.hypot(M.pts[i].x - M.pts[j].x, M.pts[i].y - M.pts[j].y));
    }
    check(min > M.tileR * 1.9, `${m.id} tiles too close: ${min.toFixed(1)} vs ${M.tileR.toFixed(1)}`);
    // special tiles never overlap
    for (const b of M.boost) { check(!M.safe.has(b), `${m.id} boost on safe ${b}`); check(!M.junction.has(b), `${m.id} boost on junction`); check(!M.entry.includes(b), `${m.id} boost on gate`); }
    for (const s of M.safe) { check(!M.junction.has(s), `${m.id} safe on junction`); check(!M.entry.includes(s) || false, `${m.id} safe on gate ${s}`); }
    // gate differs from start
    check(M.entry[0] !== M.start[0] && M.entry[1] !== M.start[1], `${m.id} gate == start`);
    // lanes do not cross each other and stay clear of the centre
    const h0 = M.lanes[0][M.lane - 1], h1 = M.lanes[1][M.lane - 1];
    check(Math.hypot(h0.x - h1.x, h0.y - h1.y) > M.tileR * 3, `${m.id} homes too close`);
    // symmetry: player two is the 180deg image of player one
    const a = M.pts[M.start[0]], b = M.pts[M.start[1]];
    const sym = (u, v) => (Math.abs(u.x + v.x) < 1.5 && Math.abs(u.y + v.y) < 1.5) || (m.id === 'infinity' && (portrait ? Math.abs(u.y + v.y) < 1.5 && Math.abs(u.x - v.x) < 1.5 : Math.abs(u.x + v.x) < 1.5 && Math.abs(u.y - v.y) < 1.5));
    check(sym(a, b), `${m.id} starts not symmetric`);
    const e0 = M.pts[M.entry[0]], e1 = M.pts[M.entry[1]];
    check(sym(e0, e1), `${m.id} gates not symmetric`);
    if (!portrait) console.log(m.id.padEnd(9), 'n', M.n, 'ringLen', M.rl, 'pathLen', M.len, 'spacing', M.spacing.toFixed(1), 'minGap', min.toFixed(1), 'gate0', JSON.stringify(e0, (k, v) => typeof v === 'number' ? Math.round(v) : v));
  }
}

// full games
const levels = ['easy', 'normal', 'hard'];
for (const m of DL.Maps.list) {
  const res = { wins: [0, 0], stuck: 0, rolls: 0, caps: 0, boosts: 0, games: 0 };
  for (let g = 0; g < 360; g++) {
    const tokens = 1 + (g % 6);
    const st = DL.Engine.create({ map: m.id, tokens, bot: true, level: levels[g % 3], names: ['A', 'B'], seed: g * 7 + 1, portrait: g % 2 === 0 });
    let done = false;
    for (let step = 0; step < 30000; step++) {
      const p = st.turn;
      const r = DL.Engine.rollDice(st, p);
      let extra = false;
      if (r.legal.length) {
        const ti = DL.Engine.botChoose(st, p, r.v, r.legal);
        const plan = DL.Engine.doMove(st, p, ti, r.v);
        if (plan.boosted) res.boosts++;
        // invariant: no token past home
        check(st.players[p].tokens.every((t) => t <= st.map.home), 'token beyond home');
        if (DL.Engine.isWin(st, p)) { res.wins[p]++; done = true; break; }
        extra = r.v === 6 || plan.captured || plan.reachedHome;
      }
      if (!extra) st.turn = 1 - st.turn;
    }
    if (!done) res.stuck++;
    res.games++;
    res.rolls += st.players[0].stats.rolls + st.players[1].stats.rolls;
    res.caps += st.players[0].stats.captures + st.players[1].stats.captures;
  }
  console.log(m.id.padEnd(9), 'wins', res.wins.join('/'), 'stuck', res.stuck, 'avgRolls', Math.round(res.rolls / res.games), 'avgCaps', (res.caps / res.games).toFixed(1), 'boosts', res.boosts);
  check(res.stuck === 0, m.id + ' has stuck games');
}
console.log(failed ? `\n${failed} FAILURES` : '\nall engine checks passed');
process.exit(failed ? 1 : 0);
